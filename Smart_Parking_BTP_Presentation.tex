% ==================== SLIDE 3: PROBLEM STATEMENT ====================
\begin{frame}{Problem Statement \& Motivation}
    \begin{columns}
        \begin{column}{0.62\textwidth}
            \begin{itemize}
                \item \textbf{Urban Traffic Congestion:} Up to 30\% of urban traffic is caused by drivers cruising in search of vacant parking slots (15--20 min/trip).
                \item \textbf{Carbon Emissions \& Fuel Waste:} Idle cruising contributes heavily to greenhouse gas ($CO_2$) emissions and wasted fuel.
                \item \textbf{Static, Blind Allocation:} Conventional lots provide no real-time occupancy updates, leading to aisle-level bottlenecks.
                \item \textbf{Project Objective:} An end-to-end intelligent ecosystem combining IoT sensing, predictive AI, shortest-path routing, and an interactive mobile app.
            \end{itemize}
        \end{column}
        \begin{column}{0.38\textwidth}
            \begin{center}

% ==================== SLIDE 6: AI CORE ====================
\begin{frame}{Artificial Intelligence Core: MARL, GNN \& GRU}
    \begin{block}{1. Graph Neural Network (GNN) Topology}
        Models the parking facility as a spatial topological graph. Replaces Euclidean distance with real physical road network traversal distances.
    \end{block}
    
    \begin{block}{2. Gated Recurrent Unit (GRU) Temporal Predictor}
        Evaluates temporal occupancy sequences to forecast whether an available slot will remain vacant by the time the driver physically arrives.
    \end{block}
    
    \begin{block}{3. Multi-Agent Reinforcement Learning (MARL) Allocation}
        Agents coordinate slot assignment by maximizing a multi-objective reward function:
        \vspace{-0.1cm}
        \[
            R(s_i) = w_1 \cdot \text{Availability}(s_i) - w_2 \cdot \text{GraphDistance}(s_i) - w_3 \cdot \text{CongestionPenalty}
        \]
    \end{block}
\end{frame}

% ==================== SLIDE 7: ROUTE OPTIMIZATION (OUR WORK) ====================
\begin{frame}{Route Optimization \& Internal Navigation (Our Work)}
    \begin{columns}
        \begin{column}{0.5\textwidth}
            \textbf{Graph-Based Shortest Path Algorithm:}
            \begin{itemize}
                \item \textbf{Network Representation:} Nodes represent Entry, Junctions ($J_{\text{main}}, J_A, J_B$), and Slots ($A_1\dots A_3, B_1\dots B_2$); edges represent physical aisle segments with metric weights.
                \item \textbf{Dijkstra Implementation:} Computes the exact driving trajectory, preventing internal lot congestion.
                \item \textbf{Dynamic Polyline Generation:} Encodes waypoint coordinates directly into the navigation stream.
                \item \textbf{Driving Distance Input:} Real shortest distance is fed back to the MARL reward engine.
            \end{itemize}
        \end{column}
